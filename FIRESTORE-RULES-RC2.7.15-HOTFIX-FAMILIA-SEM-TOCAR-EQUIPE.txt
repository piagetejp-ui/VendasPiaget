rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    // RC2.7.15 — autenticações da equipe e da família são trilhos paralelos.
    // Claims familiares usam get(..., padrão), para tokens internos sem essas
    // claims nunca quebrarem a avaliação das regras.
    function isFamily() {
      return signedIn()
        && request.auth.token.get('role', '') == 'responsavel'
        && request.auth.token.get('responsavelId', '') is string
        && request.auth.token.get('sessionId', '') is string
        && request.auth.token.get('alunosIds', []) is list;
    }

    function familyId() {
      return request.auth.token.get('responsavelId', '');
    }

    // Família: espelho seguro criado pelo backend antes do Custom Token.
    function familyMirrorValid() {
      return isFamily()
        && exists(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', '')))
        && get(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', ''))).data.get('ativo', false) == true
        && get(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', ''))).data.get('firebaseUid', '') == request.auth.uid
        && get(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', ''))).data.get('responsavelId', '') == familyId()
        && get(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', ''))).data.get('sessionId', '') == request.auth.token.get('sessionId', '')
        && get(/databases/$(database)/documents/familias_auth/$(request.auth.token.get('sessionId', ''))).data.get('expiraEmMs', 0) > request.time.toMillis();
    }

    function familyActive() {
      return familyMirrorValid();
    }

    function ownsStudent(studentId) {
      return familyActive()
        && studentId is string
        && request.auth.token.get('alunosIds', []).hasAny([studentId]);
    }

    function ownsFamilyDoc(data) {
      return familyActive() && (
        ('alunoId' in data && data.alunoId is string && request.auth.token.get('alunosIds', []).hasAny([data.alunoId]))
        || ('alunosIds' in data && data.alunosIds is list && request.auth.token.get('alunosIds', []).hasAny(data.alunosIds))
        || ('destinatariosAlunos' in data && data.destinatariosAlunos is list && request.auth.token.get('alunosIds', []).hasAny(data.destinatariosAlunos))
        || ('responsavelFinanceiroId' in data && data.responsavelFinanceiroId is string && data.responsavelFinanceiroId == familyId())
        || ('responsavelId' in data && data.responsavelId is string && data.responsavelId == familyId())
      );
    }

    // Equipe: caminho principal pelo espelho usuarios_auth. Não depende de
    // isFamily(), portanto um token interno sem claims familiares não bloqueia
    // a Gestão/Secretaria.
    function staffMirrorExists() {
      return signedIn()
        && exists(/databases/$(database)/documents/usuarios_auth/$(request.auth.uid));
    }

    // Fallback do piloto já validado anteriormente: confere diretamente os
    // perfis internos conhecidos pelo mesmo UID ou pelo e-mail autenticado.
    function staffKnownProfile(profileId) {
      return signedIn()
        && exists(/databases/$(database)/documents/usuarios_acesso/$(profileId))
        && get(/databases/$(database)/documents/usuarios_acesso/$(profileId)).data.get('ativo', false) == true
        && (
          get(/databases/$(database)/documents/usuarios_acesso/$(profileId)).data.get('authUid', '') == request.auth.uid
          || (
            get(/databases/$(database)/documents/usuarios_acesso/$(profileId)).data.get('email', '') != ''
            && get(/databases/$(database)/documents/usuarios_acesso/$(profileId)).data.get('email', '') == request.auth.token.get('email', '')
          )
        );
    }

    function staffActive() {
      // RC2.7.15 — guard clause exclusivamente por claim.
      // Para tokens de responsável, encerra aqui sem consultar usuarios_auth
      // ou perfis internos. Para Gestão/Secretaria, o caminho abaixo permanece
      // idêntico ao hotfix já validado no piloto.
      return signedIn()
        && request.auth.token.get('role', '') != 'responsavel'
        && (
          (
            staffMirrorExists()
            && get(/databases/$(database)/documents/usuarios_auth/$(request.auth.uid)).data.get('ativo', false) == true
          )
          || staffKnownProfile('lucas')
          || staffKnownProfile('daniele')
          || staffKnownProfile('evanda')
          || staffKnownProfile('ruan')
        );
    }

    // Equipe interna validada pelo backend no login.
    match /{document=**} {
      allow read, write: if staffActive();
    }

    // Dados necessários para a experiência do Meu Piaget.
    match /configuracoes/{id} {
      allow read: if familyActive() && id == 'sistema';
    }
    match /produtos/{id} {
      allow read: if familyActive();
    }
    match /catalogo_categorias/{id} {
      allow read: if familyActive();
    }
    match /catalogo_itens/{id} {
      allow read: if familyActive();
    }
    match /modelos_farda/{id} {
      allow read: if familyActive();
    }
    match /estoques/{id} {
      allow read: if familyActive();
    }
    match /disponibilidade_salgados/{id} {
      allow read: if familyActive();
    }

    // O responsável só enxerga os próprios alunos vinculados.
    match /alunos/{studentId} {
      allow read: if ownsStudent(studentId);
    }

    // Conta financeira é somente leitura no navegador do responsável.
    // Autorizações/limite passam pela API, que recalcula bloqueios no servidor.
    match /contas_responsaveis/{responsavelId} {
      allow read: if familyActive() && responsavelId == familyId();
    }

    // Dados do comprador são privados da própria família/aluno.
    match /dados_pagamento_responsavel/{studentId} {
      allow read: if ownsStudent(studentId);
      allow create: if ownsStudent(studentId)
        && request.resource.data.alunoId == studentId
        && request.resource.data.keys().hasOnly([
          'alunoId','alunoNome','matricula','customer','atualizadoEm',
          'atualizadoPorId','atualizadoPorNome','atualizadoPorPerfil'
        ]);
      allow update: if ownsStudent(studentId)
        && request.resource.data.alunoId == studentId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'alunoId','alunoNome','matricula','customer','atualizadoEm',
          'atualizadoPorId','atualizadoPorNome','atualizadoPorPerfil'
        ]);
    }

    // Registros operacionais: leitura somente quando pertencem à família.
    match /movimentos_conta/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pagamentos_checkout/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /vendas/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /vendas_online_links/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos_farda/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos_operacionais/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /ocorrencias_entrega/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }

    // Notificações: família pode ler as próprias e alterar somente leitura.
    match /notificacoes/{id} {
      allow read: if ownsFamilyDoc(resource.data);
      allow update: if ownsFamilyDoc(resource.data)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'lidoPor','lidoEm','atualizadoEm'
        ]);
    }

    // Nenhum outro acesso direto é liberado ao responsável.
  }
}
