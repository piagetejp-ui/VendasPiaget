rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isFamily() {
      return signedIn()
        && request.auth.token.role == 'responsavel'
        && request.auth.token.responsavelId is string
        && request.auth.token.sessionId is string
        && request.auth.token.alunosIds is list;
    }

    function familyId() {
      return request.auth.token.responsavelId;
    }

    function familySessionValid() {
      return isFamily()
        && exists(/databases/$(database)/documents/sessoes_meu_piaget/$(request.auth.token.sessionId))
        && get(/databases/$(database)/documents/sessoes_meu_piaget/$(request.auth.token.sessionId)).data.status == 'ativa'
        && get(/databases/$(database)/documents/sessoes_meu_piaget/$(request.auth.token.sessionId)).data.responsavelId == familyId()
        && get(/databases/$(database)/documents/sessoes_meu_piaget/$(request.auth.token.sessionId)).data.expiraEmTs > request.time;
    }

    function familyActive() {
      return familySessionValid()
        && exists(/databases/$(database)/documents/responsaveis_acesso/$(familyId()))
        && get(/databases/$(database)/documents/responsaveis_acesso/$(familyId())).data.ativo == true;
    }

    function ownsStudent(studentId) {
      return familyActive()
        && studentId is string
        && request.auth.token.alunosIds.hasAny([studentId]);
    }

    function ownsFamilyDoc(data) {
      return familyActive() && (
        ('alunoId' in data && data.alunoId is string && request.auth.token.alunosIds.hasAny([data.alunoId]))
        || ('alunosIds' in data && data.alunosIds is list && request.auth.token.alunosIds.hasAny(data.alunosIds))
        || ('destinatariosAlunos' in data && data.destinatariosAlunos is list && request.auth.token.alunosIds.hasAny(data.destinatariosAlunos))
        || ('responsavelFinanceiroId' in data && data.responsavelFinanceiroId is string && data.responsavelFinanceiroId == familyId())
        || ('responsavelId' in data && data.responsavelId is string && data.responsavelId == familyId())
      );
    }

    function staffMirrorExists() {
      return signedIn()
        && !isFamily()
        && exists(/databases/$(database)/documents/usuarios_auth/$(request.auth.uid));
    }

    function staffActive() {
      // O espelho usuarios_auth só é criado/atualizado pelo backend após
      // validar o Firebase ID Token contra usuarios_acesso.
      return staffMirrorExists()
        && get(/databases/$(database)/documents/usuarios_auth/$(request.auth.uid)).data.ativo == true;
    }

    // Equipe interna validada pelo backend no login.
    match /{document=**} {
      allow read, write: if staffActive();
    }

    // Dados necessários para a experiência do Meu Piaget.
    match /configuracoes/{id} {
      allow read: if familyActive() && id == 'sistema';
    }
    match /produtos/{id} {
      allow read: if familyActive();
    }
    match /catalogo_categorias/{id} {
      allow read: if familyActive();
    }
    match /catalogo_itens/{id} {
      allow read: if familyActive();
    }
    match /modelos_farda/{id} {
      allow read: if familyActive();
    }
    match /estoques/{id} {
      allow read: if familyActive();
    }
    match /disponibilidade_salgados/{id} {
      allow read: if familyActive();
    }

    // O responsável só enxerga os próprios alunos vinculados.
    match /alunos/{studentId} {
      allow read: if ownsStudent(studentId);
    }

    // Conta financeira é somente leitura no navegador do responsável.
    // Autorizações/limite passam pela API, que recalcula bloqueios no servidor.
    match /contas_responsaveis/{responsavelId} {
      allow read: if familyActive() && responsavelId == familyId();
    }

    // Dados do comprador são privados da própria família/aluno.
    match /dados_pagamento_responsavel/{studentId} {
      allow read: if ownsStudent(studentId);
      allow create: if ownsStudent(studentId)
        && request.resource.data.alunoId == studentId
        && request.resource.data.keys().hasOnly([
          'alunoId','alunoNome','matricula','customer','atualizadoEm',
          'atualizadoPorId','atualizadoPorNome','atualizadoPorPerfil'
        ]);
      allow update: if ownsStudent(studentId)
        && request.resource.data.alunoId == studentId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'alunoId','alunoNome','matricula','customer','atualizadoEm',
          'atualizadoPorId','atualizadoPorNome','atualizadoPorPerfil'
        ]);
    }

    // Registros operacionais: leitura somente quando pertencem à família.
    match /movimentos_conta/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pagamentos_checkout/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /vendas/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /vendas_online_links/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos_farda/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /pedidos_operacionais/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }
    match /ocorrencias_entrega/{id} {
      allow read: if ownsFamilyDoc(resource.data);
    }

    // Notificações: família pode ler as próprias e alterar somente leitura.
    match /notificacoes/{id} {
      allow read: if ownsFamilyDoc(resource.data);
      allow update: if ownsFamilyDoc(resource.data)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'lidoPor','lidoEm','atualizadoEm'
        ]);
    }

    // Nenhum outro acesso direto é liberado ao responsável.
  }
}
